# Instalăm pdfplumber !pip install pdfplumber -q import pdfplumber import csv import re from google.colab import files print("📂 PASUL 1: Încarcă fișierul PDF (dictionarul original.pdf)...") uploaded = files.upload() if not uploaded: print("❌ Nu a fost încărcat niciun fișier.") else: pdf_filename = list(uploaded.keys())[0] print(f"\n✅ Fișierul '{pdf_filename}' încărcat.") print("⚡ Pornim extragerea avansată (cu reparare bug font și detecție coloane)...") output_csv = 'dex_final_30k.csv' count = 0 # Harta pentru eliminarea accentelor de intensitate accent_map = str.maketrans('áéíóúÁÉÍÓÚ', 'aeiouAEIOU') # Lista de tag-uri gramaticale pentru a identifica startul unei intrări grammar_tags = r'(?:s\.[mnf]\.|adj\.|adv\.|vb\.[IVX]*|prep\.|conj\.|interj\.|num\.|art\.|loc\.|expr\.|fig\.|pop\.|fam\.|rar|livr\.|reg\.|argou|invar\.|tr\.|intr\.|refl\.|impers\.|aux\.|copul\.|şi|sau|,|num\.card\.|num\.ord\.|pron\.|pron\.interog\.|pron\.nehot\.|pron\.dem\.|pron\.pos\.|pron\.refl\.|art\.hot\.|art\.nehot\.|loc\.adv\.|loc\.prep\.|loc\.conj\.|vb\.[IVX]+\s+tr\.|vb\.[IVX]+\s+intr\.|vb\.[IVX]+\s+refl\.)' # Regex pentru o intrare nouă: Termen + Gramatică + Definiție # Termenul poate conține litere, cifre, cratime, tilde entry_pattern = re.compile( r'^' r'([A-Za-z0-9ĂÂÎȘăâîșț\-~]+(?:,\s*‐?[ăâîșța-zA-Z]+)?)' # Grup 1: Termen r'\s+' r'(' + grammar_tags + r'+)' # Grup 2: Gramatică r'\s*' r'(.+)$', # Grup 3: Definiție re.IGNORECASE ) def clean_term(raw_term): """Curăță termenul de bug-ul fontului (Ț inserat) și alte artefacte.""" # 1. Eliminăm bug-ul specific: 'Ț' inserat în fața fiecărei litere # Ex: "ȚEȚXȚC" -> "EXC" cleaned = re.sub(r'Ț([A-Za-z])', r'\1', raw_term) # Dacă mai sunt Ț-uri izolate la început, le eliminăm cleaned = cleaned.strip('Ț') # 2. Eliminăm accentul de intensitate și transformăm în majuscule cleaned = cleaned.translate(accent_map).upper() # 3. Corectăm sedila (veche) în virgulă (nouă) cleaned = cleaned.replace('Ţ', 'Ț').replace('Ş', 'Ș').replace('ţ', 'ț').replace('ş', 'ș') # 4. Păstrăm doar literele românești, cifrele și cratimele cleaned = re.sub(r'[^A-ZĂÂÎȘȚ0-9\-]', '', cleaned) return cleaned with open(output_csv, 'w', encoding='utf-8', newline='') as f: writer = csv.writer(f) writer.writerow(['termen', 'definitie', 'taguri']) current_entry = None try: with pdfplumber.open(pdf_filename) as pdf: total_pages = len(pdf.pages) print(f" PDF-ul are {total_pages} pagini. Începem...\n") for i, page in enumerate(pdf.pages): # Folosim layout=True pentru a păstra structura vizuală a coloanelor text = page.extract_text(layout=True) if not text: continue # Procesăm fiecare linie lines = text.split('\n') for line in lines: # Separăm coloanele: dacă linia are un spațiu mare (3+ spații), o împărțim # Acest lucru separă textul din coloana 1 de cel din coloana 2 segments = re.split(r'\s{3,}', line) for segment in segments: segment = segment.strip() if not segment: continue match = entry_pattern.match(segment) if match: raw_term = match.group(1).strip() raw_gram = match.group(2).strip() raw_def = match.group(3).strip() clean_term_text = clean_term(raw_term) if clean_term_text and len(clean_term_text) >= 2: # Salvăm intrarea anterioară if current_entry: writer.writerow([current_entry['termen'], current_entry['definitie'], current_entry['taguri']]) count += 1 # Generăm tag-uri taguri = ["dex", "romana"] gr_lower = raw_gram.lower() if "s.m." in gr_lower or "s.f." in gr_lower or "s.n." in gr_lower: taguri.append("substantiv") elif "adj" in gr_lower: taguri.append("adjectiv") elif "vb." in gr_lower or "tr." in gr_lower or "intr." in gr_lower or "refl." in gr_lower: taguri.append("verb") elif "adv" in gr_lower: taguri.append("adverb") elif "interj" in gr_lower: taguri.append("interjecție") elif "prep" in gr_lower: taguri.append("prepoziție") elif "conj" in gr_lower: taguri.append("conjuncție") elif "pron" in gr_lower: taguri.append("pronume") elif "num" in gr_lower: taguri.append("numeral") elif "art" in gr_lower: taguri.append("articol") elif "loc" in gr_lower or "expr" in gr_lower: taguri.append("expresie") current_entry = { 'termen': clean_term_text, 'definitie': raw_def.replace('
', ' ').replace('\n', ' ').strip(), 'taguri': ", ".join(taguri) } else: # Linii de continuare a definiției if current_entry: clean_line = segment.replace('
', ' ').replace('|', ' ').strip() # Ignorăm etimologia și indicațiile de final ignore_patterns = [ r'^–\s*Sil\.', r'^–\s*Pl\.', r'^–\s*G\.‐D\.', r'^–\s*Acc\.', r'^–\s*Pr\.', r'^–\s*Scris', r'^–\s*Var\.', r'^–\s*Par\.', r'^–\s*Forma', r'^\d+\.?\s*$' ] should_ignore = any(re.match(p, clean_line) for p in ignore_patterns) if clean_line and not should_ignore and len(clean_line) > 5: current_entry['definitie'] += ' ' + clean_line if (i + 1) % 50 == 0: print(f"🔄 Progres: Pagina {i+1} din {total_pages}... (Găsite {count} definiții)") except Exception as e: print(f"❌ Eroare majoră: {e}") # Salvăm ultima intrare if current_entry: writer.writerow([current_entry['termen'], current_entry['definitie'], current_entry['taguri']]) count += 1 print(f"\n🎉 SUCCES! Am procesat {count} de definiții.") print(f"📁 Fișierul a fost salvat ca: {output_csv}") # Afișăm primele 15 linii pentru verificare print("\n🔍 PRIMELE 15 LINII DIN CSV (verifică dacă termenii sunt curați):") with open(output_csv, 'r', encoding='utf-8') as f: for idx, line in enumerate(f): if idx < 16: print(f" {idx}: {line.strip()[:100]}") print("\n⬇️ Descărcarea fișierului începe AUTOMAT...") files.download(output_csv)
